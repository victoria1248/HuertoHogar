esValido = false; } else if (password.length < 8) { if (errPassword) errPassword.textContent = 'Debe tener al menos 8 caracteres.'; esValido = false; } // Validar Confirmación if (confirmPassword !== password) { if (errConfirm) errConfirm.textContent = 'Las contraseñas no coinciden.'; const errCorreo = document.getElementById('err-admin-correo'); const errPass = document.getElementById('err-admin-password'); const alertaError = document.getElementById('admin-error-general'); if (errCorreo) errCorreo.textContent = ''; const password = adminPassInput ? adminPassInput.value : ''; let esValido = true; // Validar correo requerido y dominio if (correo === '') { if (errCorreo) errCorreo.textContent = 'Ingrese su correo de administración.'; esValido = false; } else { const correoValido = DOMINIOS_PERMITIDOS.some(dominio => correo.toLowerCase().endsWith(dominio) ); if (!correoValido) { if (errCorreo) errCorreo.textContent = 'El correo debe utilizar un dominio válido (@gmail.com, @duocuc.cl, etc.).'; esValido = false; } } // Validar clave requerida if (password === '') { if (errPass) errPass.textContent = 'Ingrese su contraseña de administración.'; esValido = false; } if (!esValido) return; // Cargar administradores de localStorage let adminsGuardados = []; } return; } // Credenciales por defecto o registradas const esAdminPrincipal = (correo.toLowerCase() === 'admin@huertohogar.cl' && password === 'Admin123!'); const esAdminSecundario = (correo.toLowerCase() === 'admin@gmail.com' && password === 'Admin123!'); } catch (err) {} window.location.href = 'dashbord.html'; } else { if (alertaError) { alertaError.classList.remove('oculto'); alertaError.textContent = 'Correo o contraseña de administrador incorrectos o no autorizados.'; } } }); } });
